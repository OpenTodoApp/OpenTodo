#include <tunables/global>

# AppArmor profile for the OpenTodo add-on: the s6 init and run.sh may run what they need,
# and OpenTodo itself runs in its own profile that may only read its binary, a few system
# files and write its data in /data.
profile opentodo flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  file,
  signal (send) set=(kill,term,int,hup,cont),
  capability chown,
  capability dac_override,
  capability fowner,
  capability setgid,
  capability setuid,

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio and the start script
  /usr/lib/bashio/** ix,
  /tmp/** rwk,
  /run.sh rix,

  # The add-on's options and OpenTodo's data
  /data/** rw,

  # OpenTodo runs in its own profile
  /usr/bin/opentodo cx -> opentodo,

  profile opentodo flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>
    #include <abstractions/ssl_certs>

    # Receive signals from S6-Overlay and Docker
    signal (receive) peer=*_opentodo,
    signal (receive) peer=unconfined,

    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network netlink raw,

    /usr/bin/opentodo rm,
    /data/ r,
    /data/** rwkl,
    /tmp/** rwk,
    /etc/passwd r,
    /etc/group r,
    /etc/hosts r,
    /etc/resolv.conf r,
    /etc/ssl/** r,
    /etc/localtime r,
    /usr/share/zoneinfo/** r,
    /proc/sys/net/core/somaxconn r,
    /sys/kernel/mm/transparent_hugepage/hpage_pmd_size r,
    /dev/tty rw,
  }
}
